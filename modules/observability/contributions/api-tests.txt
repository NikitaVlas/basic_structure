test("observability middleware correlates requests and exposes bounded metrics", async () => {
  const server = createAppServer().listen(0);
  await new Promise((resolve) => server.once("listening", resolve));
  const address = server.address();
  assert.ok(address && typeof address === "object");
  try {
    const health = await fetch(`http://127.0.0.1:${address.port}/health`, { headers: { "x-request-id": "smoke-request-123" } });
    assert.equal(health.headers.get("x-request-id"), "smoke-request-123");
    const metrics = await fetch(`http://127.0.0.1:${address.port}/metrics`);
    const body = await metrics.text();
    assert.match(metrics.headers.get("content-type") ?? "", /text\/plain/);
    assert.match(body, /http_requests_total\{method="GET",route="\/health",status="200"\} [1-9][0-9]*/);
    assert.doesNotMatch(body, /smoke-request-123/);
  } finally {
    await new Promise<void>((resolve, reject) => server.close((error) => error ? reject(error) : resolve()));
  }
});
