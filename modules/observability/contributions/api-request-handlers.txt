const requestId = requestIdFrom(request.headers["x-request-id"]);
request.headers["x-request-id"] = requestId;
response.setHeader("x-request-id", requestId);
const requestStarted = process.hrtime.bigint();
const route = observableRoute(requestUrl.pathname);
httpActive.add(1);
response.once("finish", () => {
  const durationSeconds = Number(process.hrtime.bigint() - requestStarted) / 1_000_000_000;
  const labels = { method: request.method ?? "UNKNOWN", route, status: String(response.statusCode) };
  httpActive.add(-1);
  httpRequests.add(1, labels);
  httpDurationCount.add(1, labels);
  httpDurationSum.add(durationSeconds, labels);
  apiLogger.child({ requestId }).log(response.statusCode >= 500 ? "error" : response.statusCode >= 400 ? "warn" : "info", "http_request_completed", { ...labels, durationMs: Math.round(durationSeconds * 1000) });
});
if (request.method === "GET" && requestUrl.pathname === "/metrics") {
  response.setHeader("content-type", "text/plain; version=0.0.4; charset=utf-8");
  response.writeHead(200).end(apiMetrics.render());
  return;
}
